\begin{tikzpicture}[
    >=Stealth,
    node distance=0.9cm and 0.9cm,
    input/.style={
        draw=black!60,
        line width=0.75pt,
        rounded corners=2pt,
        fill=black!3,
        align=center,
        minimum height=1.12cm,
        text width=2.45cm,
        inner sep=6pt,
        font=\small
    },
    proc/.style={
        draw=WaloFigBlue,
        line width=0.8pt,
        rounded corners=2pt,
        fill=WaloFigBlueLight,
        align=center,
        minimum height=1.12cm,
        text width=2.65cm,
        inner sep=6pt,
        font=\small
    },
    fftstyle/.style={
        draw=WaloFigBlue,
        line width=0.85pt,
        rounded corners=2pt,
        fill=WaloFigBlueLight,
        align=center,
        minimum height=1.12cm,
        text width=2.75cm,
        inner sep=6pt,
        font=\small
    },
    outnode/.style={
        draw=WaloFigOrange,
        line width=0.9pt,
        rounded corners=2pt,
        fill=WaloFigOrangeLight,
        align=center,
        minimum height=1.15cm,
        text width=3.0cm,
        inner sep=7pt,
        font=\small
    },
    arrow/.style={->, line width=0.8pt, draw=black!68}
]

% Fila superior: adquisición, ventana, FFT y extracción de magnitud/fase.
\node[input] (block) {
    Bloque de muestras\\
    $v_i[n]$\\
    $n=0,\ldots,N-1$
};

\node[proc, right=of block] (window) {
    Enventanado\\
    $x_i[n]=w[n]v_i[n]$\\
    Blackman--Harris
};

\node[fftstyle, right=of window] (fft) {
    Transformada discreta\\
    de Fourier\\
    $Z_i[k]$ (FFT)
};

\node[proc, right=of fft] (polar) {
    Magnitud y fase\\
    $|Z_i[k]|$ y $\Phi_i[k]$\\
    Cálculo CORDIC
};

\draw[arrow] (block) -- (window);
\draw[arrow] (window) -- (fft);
\draw[arrow] (fft) -- (polar);

% Fila inferior: lectura de derecha a izquierda, con mayor separación vertical.
\node[proc, below=1.65cm of polar] (peak) {
    Identificación del\\
    máximo espectral\\
    Bin $k_0$
};

\node[proc, left=of peak, text width=2.85cm] (interp) {
    Interpolación\\
    parabólica local\\
    Desplazamiento $\delta$
};

\node[outnode, left=of interp] (freq) {
    \textbf{Frecuencia estimada}\\
    $\widehat{f}_0=(k_0+\delta)\dfrac{f_s'}{N}$
};

\draw[arrow] (polar.south) -- (peak.north);
\draw[arrow] (peak) -- (interp);
\draw[arrow] (interp) -- (freq);

% Resolución espectral como nota secundaria, fuera del flujo de flechas.
\node[
    font=\footnotesize,
    text=black!65,
    align=center,
    below=0.38cm of interp
] (resolution) {
    Separación entre bines: $\Delta f=f_s'/N$
};

\end{tikzpicture}
